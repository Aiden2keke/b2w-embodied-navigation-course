cmake_minimum_required(VERSION 3.16)
project(b2w_contact_observer)
set(CMAKE_CXX_STANDARD 17)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
find_package(ignition-transport11 REQUIRED)
add_library(b2w_contact_observer SHARED observer.cpp)
target_link_libraries(b2w_contact_observer PRIVATE ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register ignition-transport11::core)
add_executable(test_observer test_observer.cpp)
target_compile_options(test_observer PRIVATE -UNDEBUG)
target_link_libraries(test_observer PRIVATE ignition-gazebo6::ignition-gazebo6)
enable_testing()
add_test(NAME inventory_and_contacts COMMAND test_observer)
