cmake_minimum_required(VERSION 3.16)
project(stage4_startup_diagnostics)
set(CMAKE_CXX_STANDARD 17)
find_package(GTSAM REQUIRED)
find_package(TBB REQUIRED)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros_gz_bridge REQUIRED)
find_package(ros_gz_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-transport11 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
add_executable(gtsam_probe gtsam_probe.cpp)
target_include_directories(gtsam_probe PRIVATE ${GTSAM_INCLUDE_DIR} /usr/include/eigen3)
target_link_libraries(gtsam_probe PRIVATE gtsam TBB::tbb)
add_executable(bridge_contract bridge_contract.cpp)
ament_target_dependencies(bridge_contract rclcpp ros_gz_bridge ros_gz_interfaces std_msgs)
target_link_libraries(bridge_contract ignition-gazebo6::ignition-gazebo6 ignition-transport11::core ignition-plugin1::register)
enable_testing()
add_test(NAME valid_initial_priors_and_positive_dt COMMAND gtsam_probe CHOLESKY .005)
add_test(NAME valid_initial_priors_and_positive_dt_qr COMMAND gtsam_probe QR .005)
add_test(NAME zero_dt_reproduces_failure COMMAND gtsam_probe CHOLESKY 0)
set_tests_properties(zero_dt_reproduces_failure PROPERTIES WILL_FAIL TRUE)
add_test(NAME bridge_factory_conversion_and_native_advertisements COMMAND bridge_contract)
set_tests_properties(bridge_factory_conversion_and_native_advertisements PROPERTIES TIMEOUT 5 ENVIRONMENT "IGN_PARTITION=b2w-stage4-offline-contract-20261021")

add_library(stage4_contact_observer SHARED ../contact_observer/observer.cpp)
target_link_libraries(stage4_contact_observer PRIVATE ignition-gazebo6::ignition-gazebo6 ignition-transport11::core ignition-plugin1::register)
