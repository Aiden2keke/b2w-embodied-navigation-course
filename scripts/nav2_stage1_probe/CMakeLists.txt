cmake_minimum_required(VERSION 3.16)
project(nav2_stage1_probe)
find_package(ament_cmake REQUIRED)
find_package(nav2_core REQUIRED)
find_package(dwb_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(nav2_stage1_probe probe.cpp)
target_compile_features(nav2_stage1_probe PRIVATE cxx_std_17)
target_compile_options(nav2_stage1_probe PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(nav2_stage1_probe nav2_core dwb_core pluginlib rclcpp)
